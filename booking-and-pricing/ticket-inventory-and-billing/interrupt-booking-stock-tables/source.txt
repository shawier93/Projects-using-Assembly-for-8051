ORG 00H
JMP MAIN

ORG 0003H 
JMP EX0ISR

ORG 0030H
MAIN:
MOV 50H,#0 ;Amount need to be paid
MOV 31H,#0 ;Available Seats for gold
MOV 32H,#0 ;Price for Gold
MOV 41H,#0 ;Available Seats for silver
MOV 42H,#0 ;Price for silver
MOV DPTR,#GOLD
CLR A
MOVC A,@A+DPTR
MOV 31H,A
INC DPTR
CLR A
MOVC A,@A+DPTR
MOV 32H,A
MOV DPTR,#SILVER
CLR A
MOVC A,@A+DPTR
MOV 41H,A
INC DPTR
CLR A
MOVC A,@A+DPTR
MOV 42H,A

ACALL INLCD	
MOV A,#80H 
ACALL CMD
MOV DPTR,#STR1
ACALL WSTR
MOV A,#0C0H 
ACALL CMD
MOV DPTR,#STRINT
ACALL WSTR

MOV IE,#10000001B  
SETB IT0
CLR P3.6 ; clear ADC WR line
SETB P3.6 ; then set it - this results in the required positive edge to start a conversion

NEXTBOOKING:
ACALL CLEARLCD
MOV A,#80H 
ACALL CMD
MOV DPTR,#MENU1
ACALL WSTR
ACALL KEYPAD ;Seat Category (1.Gold,2.Silver)
CLR C
SUBB A,#30H
MOV 30H,A
CJNE A,#1,CH1
SJMP CONT

CH1:

  CJNE A,#2,CH2

  SJMP CONT

CH2:

  CJNE A,#3,CH3

  SJMP CONT

CH3:

  ACALL SUBERR

  SJMP NEXTBOOKING

CONT:
ACALL CLEARLCD
MOV A,#80H 
ACALL CMD
MOV DPTR,#STRTICKET
ACALL WSTR
ACALL KEYPAD ;Number of Tickets (Max 9 Tickets)
CLR C
SUBB A,#30H
MOV 40H,A

ACALL CLEARLCD
MOV A,30H
CJNE A,#1,NEXT1 ; If = 1, we go to check if = 2
MOV A,31H
CLR C
SUBB A,40H ; Check for available seats
JC NOAVAIL 
MOV 31H,A ; Adjust available seats
MOV B,32H  
SJMP CALC

NEXT1:
CJNE A,#2,ERROR ; if = 2 then error
CLR C
MOV A,41H
SUBB A,40H ; Check for available seats
JC NOAVAIL
MOV 41H,A
MOV B,42H ; price for silver
SJMP CALC

ERROR:
ACALL CLEARLCD
MOV A,#80H 
ACALL CMD
MOV DPTR,#STRERR
MOV R7,#2
ERR2: ACALL TMSEC
DJNZ R7,ERR2
AJMP NEXTBOOKING

CALC:

  MOV A,40H ; Number of Seats
  MUL AB    ; Multiply by price per seat
  MOV 50H,A
  MOV DPTR,#STROK
  ACALL WSTR
  MOV B,#10
  DIV AB 
  ADD A,#30H 
  ACALL WCHR 
  MOV A,B
  ADD A,#30H
  ACALL WCHR
  MOV DPTR,#STRKD
  ACALL WSTR
  MOV R7,#5
CALC1:
  ACALL TMSEC
  DJNZ R7,CALC1
  AJMP NEXTBOOKING

NOAVAIL:
ACALL CLEARLCD
MOV A,#80H 
ACALL CMD
MOV DPTR,#NOTAV
ACALL WSTR
MOV R7,#2
NOAV3: ACALL TMSEC
DJNZ R7,NOAV3
AJMP NEXTBOOKING

CLEARLCD:
;-----clear LCD-----
MOV A,#01H  
ACALL CMD
MOV R7,#14
WAIT1:
ACALL LDELAY
DJNZ R7,WAIT1
RET

SUBERR:
ACALL CLEARLCD
MOV A,#80H 
ACALL CMD
MOV DPTR,#STRERR
ACALL WSTR
MOV R7,#2
NOAV2: ACALL TMSEC
DJNZ R7,NOAV2
RET

EX0ISR:
;---- WAIT UNTIL # IS PRESSED -----
X1:
ACALL KEYPAD  
CJNE A,#'#',X1
RETI

;-- LCD Initialization Procedure starts here ----- 
INLCD:
	CLR P1.3	

	CLR P1.7	
	CLR P1.6	
	SETB P1.5		
	CLR P1.4		 	
	SETB P1.2		 
	CLR P1.2			
	ACALL LDELAY

	SETB P1.2	
	CLR P1.2	

	SETB P1.7		
	CLR P1.6	
	CLR P1.5		 
	CLR P1.4			
	SETB P1.2		 
	CLR P1.2	
	ACALL LDELAY

	CLR P1.7		
	CLR P1.6		
	CLR P1.5		 
	CLR P1.4		 	

	SETB P1.2		 
	CLR P1.2		

	SETB P1.7		
	SETB P1.6	
	SETB P1.5	
	SETB P1.4		 

	SETB P1.2		 
	CLR P1.2		
	ACALL LDELAY

	CLR P1.7		
	CLR P1.6		
	CLR P1.5		 
	CLR P1.4		 	
	SETB P1.2		 
	CLR P1.2		

	SETB P1.6		

	SETB P1.5		
	SETB P1.2		
	CLR P1.2		
	ACALL LDELAY
	RET
;--- End  of  LCD initialization  ----- 

;- Subroutine to write character in A to the  LCD----- 
WCHR: 
	PUSH ACC
	PUSH B
   	SETB P1.3
   	MOV B,A
   	ACALL COMMON
	ACALL LDELAY
   	MOV A,B
   	SWAP A
   	ACALL COMMON
	ACALL LDELAY
	POP B
	POP ACC
   	RET

;--Subroutine to write COMMAND in A to the  LCD---- 
CMD:
   	CLR P1.3
   	MOV B,A
   	ACALL COMMON
	ACALL LDELAY
   	MOV A,B
   	SWAP A
   	ACALL COMMON
   	ACALL LDELAY
   	RET

;---Common operation for CHAR write and COMMAND write  
COMMON: 
   	ANL a,#11110000B
   	ANL P1, #00001111B
   	ORL P1,A
   	SETB P1.2
   	CLR P1.2
   	RET

;--Subroutine to write A STRING character by character-
WSTR: 
        PUSH    ACC             
CONT1:   
       CLR     A               
       MOVC    A,@A+DPTR      ; move character to A 
       JZ Exit1  
PRINT:
        ACALL    WCHR                ; call procedure to write a CHAR 
        INC DPTR                       ; get next character        
       AJMP    CONT1      ; go to CONT1  
EXIT1: POP     ACC                        ; restore A 
        RET 
  
; -------  procedure delay -------
LDELAY:	
	MOV R0, #50	
	DJNZ R0, $	
	RET

;;;;; DELAY TIME USING TIMER;;;;
TMSEC:  
	MOV   TMOD,#21H ; set timer 0 as timer MOD 1
	MOV   TL0,#0CAH  ; Timer0, low byte       
	MOV   TH0,#27H ; Timer0, high byte                          
	SETB  TR0                  ; Start timer0 
BACK:   
	JNB   TF0,BACK            ; Check timer overflow flag             
	CLR   TR0                  ; Stop timer0             
	CLR   TF0                 ; Clear timer-0 overflow flag TF0             
	RET              


;.............KEYPAD............
KEYPAD: 
       MOV P0,#01110000B  ;make bits P0.4-P0.6 input
K1:    MOV P0,#01110000B           
       MOV A,P0    ;read all col. (ensure all keys open)
       ANL A,#01110000B   	;masked unused bits
       CJNE A,#01110000B,K1 ;check til all keys released
K2:           
       MOV A,P0                ;see if any key is pressed
       ANL A,#01110000B         ;mask unused bits
       CJNE A,#01110000B,OVER1 ;key pressed, await closure
       SJMP K2                  ;check if key pressed
OVER1: 
       MOV P0,#01111110B        ;ground row 0
       MOV A,P0                 ;read all columns
       ANL A,#01110000B         ;mask unused bits
       CJNE A,#01110000B,ROW_0  ;key row 0, find the col.
       MOV P0,#01111101B        ;ground row 1
       MOV A,P0                 ;read all columns
       ANL A,#01110000B         ;mask unused bits 
       CJNE A,#01110000B,ROW_1  ;keyrow 1, find the col.
       MOV P0,#01111011B        ;ground row 2
       MOV A,P0                 ;read all columns
       ANL A,#01110000B         ;mask unused bits
       CJNE A,#01110000B,ROW_2  ;key row 2, find the col.
       MOV P0,#01110111B        ;ground row 3
       MOV A,P0                 ;read all columns
       ANL A,#01110000B         ;mask unused bits
       CJNE A,#01110000B,ROW_3  ;keyrow 3, find the col.
       LJMP K2               ;if none, false input, repeat
ROW_0: MOV DPTR,#KCODE0  ;set DPTR=start of row 0        
       SJMP FIND                ;find col. key belongs to
ROW_1: MOV DPTR,#KCODE1         ;set DPTR=start of row 1
       SJMP FIND                ;find col. key belongs to
ROW_2: MOV DPTR,#KCODE2         ;set DPTR=start of row 2 
       SJMP FIND                ;find col. key belongs to
ROW_3: MOV DPTR,#KCODE3         ;set DPTR=start of row 3

FIND:  
       RLC A			;skip D7 data (unused)
FIND1:
       RLC A                    ;see if any CY bit low
       JNC MATCH     ;if zero, get the key number        
       INC DPTR              ;point to next col. address
       SJMP FIND1               ;keep searching
MATCH: CLR A                    ;set A=0 (match is found)
       MOVC A,@A+DPTR         ;get key number from table
       RET

;ASCII LOOK-UP TABLE FOR EACH ROW
KCODE3: DB '1','2','3' ;Row0 ASCII codes
KCODE2: DB '4','5','6' ;Row1
KCODE1: DB '7','8','9' ;Row2
KCODE0: DB '*','0','#' ;Row3

GOLD: DB 9,5
SILVER: DB 9,3

STR1: DB "Welcome"
      DB 0

MENU1: DB "1.G   2.S  3.E "
      DB 0
STRTICKET: DB "Tickets:"
      DB 0
STROK: DB "OK"
      DB 0
STRKD: DB "KD"
      DB 0
STRINT: DB "Press # to start"
      DB 0
NOTAV: DB "NOT AVAILABLE"
      DB 0
STRERR: DB "**INVALID**"
      DB 0

mov R0,#61H
cjne @R0,#0h,else
mov 61h,#1
