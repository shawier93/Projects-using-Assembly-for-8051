ORG 00H

MOV R0,#20H   ; STARTING SEQUANCE

MENU:
MOV 10H,#2    ; CHOICE

MOV A,10H ;SELECTION

FUNCTION1:
CJNE A,#1,FUNCTION2
ACALL OPTION1
SJMP MENU

FUNCTION2:
CJNE A,#2,FUNCTION3
ACALL OPTION2
SJMP MENU

FUNCTION3:
CJNE A,#3,WRONG
;EXIT
AJMP FINISH

WRONG:
SJMP MENU


OPTION1:
MOV R7,#6    ; N
MOV A,#3
MOV @R0,A
INC R0
MOV A,#0
MOV @R0,A
INC R0
MOV A,#2
MOV @R0,A


MOV R0,#20H
LOOP:
MOV A,@R0
INC R0
MOV B,@R0
INC R0
INC R0
ADD A,B
MOV @R0,A
DEC R0
DEC R0
DJNZ R7,LOOP
RET

OPTION2:
MOV 30H,#1    ; N
MOV R1,#30H   ; NTH VALUE 

MOV A,30H
SUBB A,#3
MOV R7,A
MOV A,#4
MOV @R0,A
INC R0
MOV A,#2
MOV @R0,A
INC R0
MOV A,#3
MOV @R0,A
ACALL CHECK

MOV R0,#20H
LOOP2:
MOV A,@R0
INC R0
MOV B,@R0
INC R0
INC R0
ADD A,B
MOV @R0,A
DEC R0
DEC R0
DJNZ R7,LOOP2
MOV @R1,A
RET

CHECK:
MOV A,30H
CJNE A,#0,S
AJMP OPTION2
S:
CJNE A,#1,S2
MOV @R1,20H
AJMP MENU
S2:
CJNE A,#2,S3
MOV @R1,21H
AJMP MENU
S3:
CJNE A,#3,S4
MOV @R1,22H
AJMP MENU
S4: RET

FINISH:
END
