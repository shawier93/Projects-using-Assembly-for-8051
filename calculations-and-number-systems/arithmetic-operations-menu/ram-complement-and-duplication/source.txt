ORG 0H

MENU:
MOV A,#0CH

CJNE A,#0AH,BIN
ADDITION:
ACALL ADDOP
AJMP EXIT

BIN:
CJNE A,#0BH,COMP
ACALL BINR
AJMP EXIT

COMP:
CJNE A,#0CH,DUP
ACALL COMPLEMENT
AJMP EXIT

DUP:
CJNE A,#0DH,EXIT
ACALL DUPLICATE
AJMP EXIT

EXIT:
; THANK YOU
AJMP OUT

///////////OP1
ADDOP:
MOV A,#5
ACALL CHECK
MOV R1,A
MOV A,#6
ACALL CHECK
ADD A,R1

MOV B,#10
DIV AB
MOV P1,A
MOV P2,B

RET


CHECK:
CJNE A,#'#',C1
AJMP ADDOP
C1:CJNE A,#'*',C2
AJMP ADDOP
C2:CJNE A,#0AH,C3
AJMP ADDOP
C3:CJNE A,#0BH,C4
AJMP ADDOP
C4:CJNE A,#0CH,C5
AJMP ADDOP
C5:CJNE A,#0DH,C6
AJMP ADDOP
C6:RET

///////////////END OF OP1

///////////////OP2
BINR:
MOV A,#5
ACALL CHECKHEX
MOV R1,A
MOV A,#0AH
ACALL CHECKHEX

SWAP A
ADD A,R1

MOV R7,#8
MOV R0,#30H
LOOP:
RLC A
JC ONEX
ZEROX:
MOV @R0,#0
INC R0
SJMP OUTNOWX
ONEX:
MOV @R0,#1
INC R0
OUTNOWX:
DJNZ R7,LOOP
INC R0
MOV @R0,#'B'
RET

CHECKHEX:
CJNE A,#'#',M1
AJMP BINR
M1:CJNE A,#'*',M2
AJMP BINR
M2:RET

DUPLICATE:
MOV A,#5
ACALL CHECK2
MOV B,A
ADD A,B
MOV P3,A
RET


CHECK2:
CJNE A,#'#',F1
AJMP DUPLICATE
F1:CJNE A,#'*',F2
AJMP DUPLICATE
F2:CJNE A,#'A',F3
AJMP DUPLICATE
F3:CJNE A,#'B',F4
AJMP DUPLICATE
F4:CJNE A,#'C',F5
AJMP DUPLICATE
F5:CJNE A,#'D',F6
AJMP DUPLICATE

ROTATE:
CJNE A,#0,NEXTCHECK
CLR C
MOV A,R1
RLC A
MOV R1,A
SJMP OUTY
NEXTCHECK:
CJNE A,#1,COMPLEMENT
SETB C

RLC A
MOV R1,A
OUTY:
